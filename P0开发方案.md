### 2.1 `core/llmerrors.py`（新文件，纯标准库，不 import astrbot）

```python
"""LLM 调用错误分类（P0-T1）：统一错误类别，供上层决定重试/降级策略。

独立模块、不依赖 astrbot，可被 tests/test_p0.py 直接导入。
"""
from __future__ import annotations

import asyncio


class LLMCallError(Exception):
    TIMEOUT = "timeout"                    # 超时（流式空闲超时 / 总超时）
    RATE_LIMIT = "rate_limit"              # 限流 / 429 / 配额
    CONTEXT_OVERFLOW = "context_overflow"  # 上下文超限（重试无意义，应裁剪注入）
    PROVIDER = "provider_error"            # 其他供应商侧错误
    EMPTY = "empty_response"               # 调用成功但内容为空

    _KIND_CN = {
        TIMEOUT: "超时",
        RATE_LIMIT: "限流",
        CONTEXT_OVERFLOW: "上下文超限",
        PROVIDER: "供应商错误",
        EMPTY: "空响应",
    }

    def __init__(self, kind: str, detail: str = ""):
        super().__init__(f"{kind}: {detail}")
        self.kind = kind
        self.detail = detail

    def __str__(self) -> str:
        head = self._KIND_CN.get(self.kind, self.kind)
        return f"{head}（{self.kind}）" + (f": {self.detail[:200]}" if self.detail else "")


def classify_llm_error(exc: BaseException, detail: str = "") -> LLMCallError:
    """把任意异常映射为 LLMCallError；detail 可覆盖异常文本。"""
    msg = detail or f"{type(exc).__name__}: {exc}"
    low = msg.lower()
    if isinstance(exc, (asyncio.TimeoutError, TimeoutError)) \
            or "timeout" in low or "timed out" in low or "超时" in msg:
        return LLMCallError(LLMCallError.TIMEOUT, msg)
    if ("429" in low or "too many requests" in low or "quota" in low
            or "rate limit" in low or "rate_limit" in low or "限流" in msg):
        return LLMCallError(LLMCallError.RATE_LIMIT, msg)
    overflow_kw = ("context length", "maximum context", "context window",
                   "token limit", "max_tokens", "too long", "请求过长",
                   "上下文超限", "payload too large")
    if any(k in low or k in msg for k in overflow_kw):
        return LLMCallError(LLMCallError.CONTEXT_OVERFLOW, msg)
    return LLMCallError(LLMCallError.PROVIDER, msg)
```

### 2.2 `core/textclean.py`（新文件，纯标准库）

```python
"""完稿深层清洗（P0-T7）：剥离正文段落级的元信息残留。

独立模块、不依赖 astrbot。只清洗「整行命中」的元信息：
分节标题（## xxx（2/3））、独立章节标题行、剧情状态标记、括号注记、引导语。
规则取向：宁可漏判、不可误伤对白与叙述。
"""
from __future__ import annotations

import re

# ① 分节标题行：任意内容 + 末尾 (n/m) 计数，可带 markdown 井号
_RE_SECTION_HEADER = re.compile(
    r"^\s*#{0,6}\s*[^\n]{0,60}?[（(]\s*\d+\s*/\s*\d+\s*[)）]\s*$"
)
# ② 独立章节标题行：第N章 后必须紧跟一个分隔符（空格/·/：/-/—/】）。
#    中文行文不用空格，「第三章的内容…」这类叙述在「章」后紧跟汉字，不会命中。
_RE_CHAPTER_TITLE = re.compile(
    r"^\s*#{0,6}\s*(?:第[0-9零一二两三四五六七八九十百千]+\s*章|Chapter\s*\d+)"
    r"[\s·:：\-—】][^\n]{1,30}$"
)
# ③ 剧情状态标记残留（任意位置整行）
_RE_STORY_MARK = re.compile(r"^\s*【?\s*(?:STORY_END|STORY_CONTINUE)\s*】?\s*$", re.I)
# ④ 独立括号注记行（不限前 8 行）
_RE_NOTE_LINE = re.compile(r"^[（(]\s*(?:注|说明|备注|提示|小记)[：:][^）)]{0,60}[）)]\s*$")
# ⑤ 引导语整行
_RE_LEADIN_LINE = re.compile(
    r"^\s*(?:好的|好|嗯|那么|ok|OK)?[，,:：\s]*"
    r"(?:以下为正文|以下是正文|正文如下|正文开始|正文内容如下|以下内容为正文)[：:，,。\s]?\s*$"
)

_SENTENCE_END = ("。", "！", "？", "…", "!", "?", "”", "」", "』")
_QUOTE_CHARS = ("\"", "“", "”", "「", "『", "《")


def line_is_meta(line: str, isolated: bool = True) -> bool:
    """判断单行是否为元信息残留（供 scrub 清单展示复用）。
    isolated=该行独立成段（前后是空行/文件边缘）时才认章节标题。"""
    s = line.strip()
    if not s:
        return False
    if _strong_meta(s):
        return True
    if isolated and _RE_CHAPTER_TITLE.match(s):
        if not any(q in s for q in _QUOTE_CHARS) and s[-1] not in _SENTENCE_END:
            return True
    return False


def deep_clean_sections(text: str) -> str:
    """逐行剥离元信息残留并压缩多余空行。每节捕获时调用。

    两遍法：第一遍剥强模式（无需上下文）；第二遍在幸存行里剥「独立标题行」
    （上一行为空或行首即视为独立，避免被同批剥掉的引导语干扰孤立判定）。
    """
    if not text:
        return text
    # 第一遍：强模式（分节标题 / 状态标记 / 注记 / 引导语）
    kept = [ln for ln in text.split("\n") if not _strong_meta(ln)]
    # 第二遍：独立章节标题行
    out_lines: list[str] = []
    for i, ln in enumerate(kept):
        if ln.strip() and _RE_CHAPTER_TITLE.match(ln.strip()):
            prev_blank = (i == 0) or (not kept[i - 1].strip())
            s = ln.strip()
            if prev_blank and not any(q in s for q in _QUOTE_CHARS) and s[-1] not in _SENTENCE_END:
                continue
        out_lines.append(ln)
    out = "\n".join(out_lines)
    out = re.sub(r"\n{3,}", "\n\n", out)
    return out.strip()


def _strong_meta(line: str) -> bool:
    """强元信息模式：整行命中即剥，不需要孤立性佐证。"""
    s = line.strip()
    if not s:
        return False
    if _RE_STORY_MARK.match(s) or _RE_NOTE_LINE.match(s) or _RE_LEADIN_LINE.match(s):
        return True
    if _RE_SECTION_HEADER.match(s):
        # 带 # 前缀必是标题；无 # 的（n/m）结尾行要求不含引号（防误伤对白）
        return s.startswith("#") or not any(q in s for q in _QUOTE_CHARS)
    return False
```

### 2.3 `core/outlineops.py`（新文件，纯标准库：分批合并逻辑，便于单测）

```python
"""大纲分批收单的纯函数（P0-T5）：合并批次、查缺章。不依赖 astrbot。"""
from __future__ import annotations

from typing import Any


def merge_outline_batch(current: dict[str, Any], batch: dict[str, Any]) -> dict[str, Any]:
    """把一批 {meta?, chapters?} 合并进当前草稿：
    meta 浅合并（批里出现的键覆盖）；chapters 按 ch 号 upsert（同号替换、新号追加后排序）。
    返回新 dict，不改入参。"""
    merged: dict[str, Any] = {
        "meta": dict(current.get("meta") or {}),
        "chapters": [dict(c) for c in (current.get("chapters") or []) if isinstance(c, dict)],
    }
    b_meta = batch.get("meta")
    if isinstance(b_meta, dict):
        merged["meta"].update(b_meta)
    b_chs = batch.get("chapters")
    if isinstance(b_chs, list):
        by_no = {int(c["ch"]): c for c in merged["chapters"]
                 if isinstance(c, dict) and isinstance(c.get("ch"), int)}
        for c in b_chs:
            if isinstance(c, dict) and isinstance(c.get("ch"), int):
                by_no[int(c["ch"])] = c
        merged["chapters"] = [by_no[k] for k in sorted(by_no)]
    return merged


def missing_chapters(outline: dict[str, Any]) -> list[int]:
    """在已有 1..max 章范围内找出缺失的章号；无章节返回空。"""
    chs = [int(c["ch"]) for c in (outline.get("chapters") or [])
           if isinstance(c, dict) and isinstance(c.get("ch"), int)]
    if not chs:
        return []
    want = set(range(1, max(chs) + 1))
    return sorted(want - set(chs))
```

### T2 `_llm_chat` 重构：流式优先 + 空闲超时 + 非流式回退（base.py）

**整段替换 base.py:94-122 的 `_llm_chat`**，新增 4 个方法 + 1 个兼容壳：

```python
    # ── LLM 调用（P0 重构：流式优先 + 空闲超时 + 错误分类） ─────────────

    def _provider_stream_ok(self, provider) -> bool:
        """探测 provider.text_chat 是否支持 stream 参数（按 provider 类缓存）。"""
        cls = type(provider)
        if cls not in self._stream_ok_cache:
            try:
                import inspect
                ok = "stream" in inspect.signature(provider.text_chat).parameters
            except Exception:
                ok = False
            self._stream_ok_cache[cls] = ok
        return self._stream_ok_cache[cls]

    @staticmethod
    def _chunk_delta(chunk) -> str:
        """流式块 → 增量文本（只认 delta/delta_text 字段；completion_text 归 _chunk_full）。"""
        if isinstance(chunk, str):
            return chunk
        for attr in ("delta", "delta_text"):
            v = getattr(chunk, attr, None)
            if isinstance(v, str) and v:
                return v
        return ""

    @staticmethod
    def _chunk_full(chunk) -> str:
        """流式块 → 截至当前的完整文本（若实现携带的是累计值）。"""
        if isinstance(chunk, str):
            return ""
        v = getattr(chunk, "completion_text", None)
        return v if isinstance(v, str) else ""

    async def _llm_stream(self, provider, prompt: str, system_prompt: str,
                          idle: float, total: float) -> str:
        """流式消费：空闲超时 idle、总超时 total；只有正常推进就不掐。"""
        agen = provider.text_chat(
            prompt=prompt, system_prompt=system_prompt or None,
            stream=True, request_max_retries=1,
        )
        parts: list[str] = []           # 增量拼接（delta 语义）
        all_full_concat: list[str] = [] # completion_text 逐块拼接（两种语义都可能）
        last_full = ""
        got_any = False
        deadline = time.monotonic() + total
        while True:
            if time.monotonic() > deadline:
                raise LLMCallError(
                    LLMCallError.TIMEOUT,
                    f"流式总超时 >{total:.0f}s（已收约 {sum(len(p) for p in parts)} 字符后中止）",
                )
            try:
                nxt = await asyncio.wait_for(agen.__anext__(), timeout=idle)
            except StopAsyncIteration:
                break
            except asyncio.TimeoutError:
                raise LLMCallError(
                    LLMCallError.TIMEOUT,
                    f"流式空闲超时 >{idle:.0f}s（已收约 {sum(len(p) for p in parts)} 字符后中止）",
                )
            except Exception as e:
                if not got_any:
                    # 首块前失败：标记 setup 失败，允许上层回退非流式
                    raise LLMCallError(LLMCallError.PROVIDER, f"stream_setup {type(e).__name__}: {e}")
                raise classify_llm_error(e)
            got_any = True
            piece = self._chunk_delta(nxt)
            if piece:
                parts.append(piece)
            full = self._chunk_full(nxt)
            if full:
                all_full_concat.append(full)
                if len(full) > len(last_full):
                    last_full = full
        # 兼容三种流式实现语义：delta 块 / 累计 completion_text / 每块自带一小段 completion_text
        delta_text = "".join(parts)
        concat_text = "".join(all_full_concat)
        if delta_text.strip():
            best = delta_text                      # delta 语义：增量拼接即全文
        elif len(all_full_concat) <= 1:
            best = last_full                        # 单块：末块即全文
        elif len(last_full) >= 0.8 * len(concat_text):
            best = last_full                        # 累计语义：末块≈全量
        else:
            best = concat_text                      # 每块一小段：拼接还原
        if not best.strip():
            raise LLMCallError(LLMCallError.EMPTY, "流式完成但未取到任何文本")
        return best

    async def _llm_nonstream(self, provider, prompt: str, system_prompt: str,
                             total: float) -> str:
        """非流式调用（回退路径 / stream_first=false）。"""
        try:
            result = await asyncio.wait_for(
                provider.text_chat(
                    prompt=prompt,
                    system_prompt=system_prompt or None,
                    request_max_retries=2,
                ),
                timeout=total,
            )
        except asyncio.TimeoutError:
            raise LLMCallError(LLMCallError.TIMEOUT, f"非流式总超时 >{total:.0f}s")
        except Exception as e:
            raise classify_llm_error(e)
        if hasattr(result, "completion_text"):
            return result.completion_text or ""
        if isinstance(result, str):
            return result
        return str(result)

    async def _llm_chat_ex(
        self, prompt: str, system_prompt: str = "", timeout: float = 0.0
    ) -> tuple[str, Optional[LLMCallError]]:
        """统一 LLM 调用入口：返回 (文本, 错误)，二者互斥，永不抛异常。
        timeout<=0 用引擎默认 llm_total_timeout。"""
        started = time.monotonic()
        try:
            provider = self.context.get_using_provider()
        except Exception as e:
            return "", LLMCallError(LLMCallError.PROVIDER, f"获取 provider 失败: {e}")
        if not provider:
            return "", LLMCallError(LLMCallError.PROVIDER, "没有可用的 provider")

        total = float(timeout) if timeout and timeout > 0 else float(self._ENGINE["llm_total_timeout"])
        text, err = "", None
        if self._ENGINE["stream_first"] and self._provider_stream_ok(provider):
            try:
                text = await self._llm_stream(
                    provider, prompt, system_prompt,
                    idle=float(self._ENGINE["stream_idle_timeout"]), total=total,
                )
            except LLMCallError as e:
                err, text = e, ""
                # 只有「首块前的 setup 失败」（如实现不支持流）才回退非流式；
                # 超时/限流不回退，避免同一次请求双倍计费
                if e.detail.startswith("stream_setup"):
                    logger.warning(f"[沙盒] 流式通路不可用（{e}），本进程内回退非流式")
                    self._stream_ok_cache[type(provider)] = False
                    try:
                        text = await self._llm_nonstream(provider, prompt, system_prompt, total)
                        err = None
                    except LLMCallError as e2:
                        err, text = e2, ""
        else:
            try:
                text = await self._llm_nonstream(provider, prompt, system_prompt, total)
            except LLMCallError as e:
                err, text = e, ""

        dur = time.monotonic() - started
        if err is None and not text.strip():
            err = LLMCallError(LLMCallError.EMPTY, f"耗时 {dur:.0f}s")
        if err is not None:
            logger.error(f"[沙盒] LLM 调用失败 dur={dur:.0f}s in={len(prompt)}chars :: {err}")
        else:
            logger.info(f"[沙盒] LLM 调用完成 dur={dur:.0f}s in={len(prompt)}chars out={len(text)}chars")
        return text, err

    async def _llm_chat(self, prompt: str, system_prompt: str = "", timeout: float = 0.0) -> str:
        """兼容壳：老调用点继续用。失败返回空串（错误详情见日志）。新代码一律用 _llm_chat_ex。"""
        text, err = await self._llm_chat_ex(prompt, system_prompt, timeout=timeout)
        return text if err is None else ""
```

```python
            for label, p in files:
                try:
                    original = p.read_text(encoding="utf-8")
                except Exception:
                    continue
                cleaned = deep_clean_sections(original)
                if cleaned != original.strip():
                    hits.append((label, p, [ln.strip() for ln in original.split("\n")
                                            if ln.strip() and line_is_meta(ln)]))
            if not hits:
                return "扫了一遍，没有发现分节标题/状态标记等元信息残留♪"
            if not targets.strip():
                lines = [f"发现 {len(hits)} 个文件有元信息残留（列出的行将被删除）："]
                for i, (label, _, bad) in enumerate(hits, 1):
                    lines.append(f"{i}. {label}（{len(bad)} 行，如：{bad[0][:50]}）")
                lines.append("确认清洗就说「scrub 全部」，或指定编号如 targets=1,3。")
                return "\n".join(lines)
            nums = [n.strip() for n in re.split(r"[,，、\s]+", targets) if n.strip()]
            do_all = any(n in ("all", "全部", "a") for n in nums)
            idx = {int(n) for n in nums if n.isdigit()}
            done = skipped = 0
            for i, (label, p, _) in enumerate(hits, 1):
                if not (do_all or i in idx):
                    continue
                try:
                    cleaned = deep_clean_sections(p.read_text(encoding="utf-8"))
                    bak = p.with_name(p.stem + ".bak_scrub" + p.suffix)
                    if not bak.exists():
                        bak.write_text(p.read_text(encoding="utf-8"), encoding="utf-8")
                    p.write_text(cleaned + "\n", encoding="utf-8")
                    done += 1
                except Exception:
                    skipped += 1
            return f"清洗完成：{done} 个文件（已留 .bak_scrub 备份），失败 {skipped} 个。"
```

misc.py 头部补 `from ..core.textclean import deep_clean_sections, line_is_meta`。注意 `line_is_meta` 在 scrub 的清单展示里调用时没传 isolated（默认 True）——与 deep_clean_sections 内部判定一致性可接受（清单仅展示）。

## 6. 风险与回滚

| 风险 | 缓解 |
|------|------|
| provider 的 stream 参数行为随 AstrBot 版本差异（增量/累计/不支持） | 三语义取长启发式 + `stream_setup` 探测失败自动禁用并回退非流式；`stream_first=false` 一键回到旧模式 |
| 流式请求部分成功后被判超时，浪费已生成 token | 错误里带「已收约 N 字符」便于诊断；不做断点续传（收益/复杂度不匹配，P0 不做） |
| save_batch 主模型交完首批就停 | 返回文案固定提示「下一批」；指南铁律第 4 条；confirm 校验兜底缺章 |
| resume 后剧情与中断前上下文有缝 | 恢复走完整 story_context 重建（与正常写作同源），拼接处由 prev_tail 衔接机制兜底；验收项 7 专门检查 |
| scrub 误删正常行 | 只删整行命中的元信息模式；自动留 `.bak_scrub` 备份；独立标题行判定要求「独立成段 + 无引号 + 无句末标点 + 章号后紧跟分隔符」四重条件 |
| 回滚 | 7 个 commit 全部可独立 revert；配置项有默认值，删除 `_conf_schema.yaml` 也不影响启动 |

## 7. 明确不做（P0 边界）

- 流式内容的部分保存/续传（只报「已收 N 字符」）
- 人设卡 save_batch（单卡 JSON 小，非超时来源；P1 改 schema 时一并考虑）
- 大纲 revise 的分批化（修订要求未点名节点原样保留，分批风险大于收益）
- `_check_story_ended` 的调用次数优化（归 P3 beat 层，届时该函数大概率被替代）
