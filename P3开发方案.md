```python
"""注入装配与摘要分层纯函数（P3）。不依赖 astrbot。"""
from __future__ import annotations

from typing import Any

_CUT_MARK = "\n…（ injected 部分因预算截断 ）…"


def assemble_injection(parts: list[tuple[int, str, str]], budget: int) -> str:
    """按优先级装配注入文本。parts=[(priority, name, text)]，priority 数字越小越重要。
    budget≤0 → 全部拼接不截断。超限时从优先级最低的未处理项开始整项丢弃；
    若最低项是唯一项且仍超限则就地截断。返回拼好的文本（段间空行）。"""
    if budget is not None and budget <= 0:
        budget = 10 ** 9
    items = sorted((p, n, t or "") for p, n, t in parts if (t or "").strip())
    if not items:
        return ""
    kept: list[tuple[str, str]] = []
    used = 0
    for prio, name, text in items:
        need = len(text) + (2 if kept else 0)
        if used + need <= budget:
            kept.append((name, text))
            used += need
            continue
        if not kept:
            # 第一项就超预算：截断保底
            kept.append((name, text[: max(200, budget - 10)] + _CUT_MARK))
            used = budget
            continue
        # 超预算：此项丢弃；若它是剩余项里优先级最低的，标记截断事实
        rest = [(p, n) for p, n, _ in items if (n, _) not in [(k[0], k[1]) for k in kept]]
        # 简化：直接注明省略来源名
        dropped = [n for p, n, t in items if t.strip() and (n not in [k[0] for k in kept])]
        tail = f"\n…（因注入预算省略：{'、'.join(dict.fromkeys(dropped))}）…"
        base = "\n\n".join(t for _n, t in kept)
        return base + tail
    return "\n\n".join(t for _n, t in kept)


def pick_to_archive(chs: list[int], keep: int = 5, threshold: int = 8) -> list[int]:
    """未归档章号中，除最近 keep 章外全部归档；未超 threshold 返回空。"""
    uniq = sorted({int(c) for c in chs if int(c) > 0})
    if len(uniq) <= threshold:
        return []
    return uniq[:-keep]
```

```python
    def render_story_memory(self, work: str, summary_limit: int = 300,
                            focus_chars: Optional[list[str]] = None) -> str:
        """拼接作品记忆：弧线摘要（已归档）+ 近章全量摘要（未归档）+ 设定书快照。
        focus_chars 非空时设定书 characters 只详列焦点角色，其余一行带过；world 值截断 200 字。"""
        items = self.load_chapter_summaries(work)
        lines = [f"【以下记忆仅属于《{work}》，禁止引入其他作品内容】"]
        arcs = self.load_arc_summaries(work)
        if arcs:
            lines.append("——弧线摘要（早期章节归档）——")
            for a in arcs:
                lines.append(f"[第{a.get('ch_start')}~{a.get('ch_end')}章] {a.get('text', '')}")
        unarch = [it for it in items if not it.get("archived")]
        if unarch:
            lines.append("——近章摘要——")
            for it in unarch:
                s = str(it.get("summary", ""))
                if len(s) > summary_limit:
                    s = s[:summary_limit] + "…"
                lines.append(f"[第{it.get('ch')}章] {s}")
        elif not arcs:
            lines.append("（还没有章节摘要）")
        state = self.load_story_state(work)
        state_lines = self._render_state(state, focus_chars)
        if state_lines:
            lines.append("——设定书（当前状态快照）——")
            lines.extend(state_lines)
        return "\n".join(lines)

    def _render_state(self, state: dict[str, Any],
                      focus_chars: Optional[list[str]] = None) -> list[str]:
        lines: list[str] = []
        meta = state.get("meta") or {}
        if not isinstance(meta, dict):
            meta = {}
        if meta.get("title"):
            lines.append(f"标题：{meta['title']}｜当前进度：第{meta.get('current_chapter', 0)}章")
        world = state.get("world") or {}
        if isinstance(world, dict):
            for k, v in world.items():
                v = str(v)
                if len(v) > 200:
                    v = v[:200] + "…"
                lines.append(f"世界观·{k}：{v}")
        chars = state.get("characters") or []
        if isinstance(chars, list):
            focus = {str(f).strip() for f in (focus_chars or []) if str(f).strip()}
            for c in chars:
                if not isinstance(c, dict):
                    continue
                name = str(c.get("name") or "").strip()
                if focus and name and name not in focus \
                        and not any(f in name or name in f for f in focus):
                    lines.append(f"角色：{name}（非本章焦点，略）")
                    continue
                parts = [str(c.get(k, "")) for k in ("name", "状态", "关系", "秘密") if c.get(k)]
                if parts:
                    lines.append("角色：" + "｜".join(parts))
        fsh = state.get("foreshadowing") or []
        if isinstance(fsh, list):
            for f in fsh:
                if isinstance(f, dict) and f.get("desc"):
                    lines.append(f"伏笔（{f.get('status', '未收')}）：{f['desc']}")
        un = state.get("unresolved") or []
        if isinstance(un, list):
            for u in un:
                lines.append(f"未解决：{u}")
        return lines
```

```python
    async def _ensure_chapter_beats(self, sandbox: Sandbox, work: str, ch: int,
                                    instruction: str) -> list[dict]:
        """取/生成本章 beat 细纲。已有直接返回；无大纲节点或无 plot_points 返回 []。"""
        stored = sandbox.get_chapter_beats(work, ch)
        if stored:
            return stored
        outline_text = sandbox.render_chapter_outline(work, ch)
        if "（大纲里还没有这一章的节点" in outline_text or "情节节点" not in outline_text:
            return []
        prompt = build_beat_expand_prompt(chapter_outline=outline_text, instruction=instruction)
        raw = self._clean(await self._llm_chat(prompt, system_prompt="你只输出 JSON，不要输出任何其他文字。"))
        data = None
        try:
            parsed = json.loads(raw)
            if isinstance(parsed, dict):
                data = parsed
        except Exception:
            m = re.search(r"\{.*\}", raw, re.S)
            if m:
                try:
                    parsed = json.loads(m.group(0))
                    if isinstance(parsed, dict):
                        data = parsed
                except Exception:
                    pass
        beats_raw = (data or {}).get("beats")
        if not isinstance(beats_raw, list) or len(beats_raw) < 2:
            return []
        beats: list[dict] = []
        for b in beats_raw[:8]:
            if not isinstance(b, dict) or not str(b.get("scene") or "").strip():
                continue
            beats.append({
                "scene": str(b.get("scene"))[:80],
                "goal_chars": max(300, min(int(b.get("goal_chars") or 0), 6000)),
                "turn": str(b.get("turn") or "")[:60],
                "carry": str(b.get("carry") or "")[:40],
            })
        if len(beats) < 2:
            return []
        sandbox.set_chapter_beats(work, ch, beats)
        return beats

    async def _arc_compress(self, sandbox: Sandbox, work: str, chs: list[int]) -> str:
        """把待归档章节摘要压缩为一段弧线摘要；失败返回空串。"""
        items = sandbox.load_chapter_summaries(work)
        sel = [f"[第{it.get('ch')}章] {it.get('summary', '')}" for it in items
               if int(it.get("ch", 0)) in set(chs)]
        if not sel:
            return ""
        prompt = build_arc_summary_prompt(summaries="\n".join(sel))
        raw = await self._llm_chat(prompt, system_prompt="你只输出摘要正文，不要输出其他文字。")
        text = self._clean(raw)
        return text[:320] if text else ""

    async def _check_outline_diverge(self, sandbox: Sandbox, work: str, ch: int) -> str:
        """刚写完第 ch 章后，检查下一章大纲是否被正文架空。返回提示语（无偏离/无法判断返回空串）。"""
        outline = sandbox.load_outline(work)
        nxt = next((it for it in (outline.get("chapters") or [])
                    if isinstance(it, dict) and int(it.get("ch", 0)) == ch + 1), None)
        if not nxt or not (nxt.get("summary") or nxt.get("plot_points")):
            return ""
        items = sandbox.load_chapter_summaries(work)
        cur = next((it for it in items if int(it.get("ch", 0)) == ch), None)
        if not cur or not str(cur.get("summary") or "").strip():
            return ""
        nxt_text = f"第{ch + 1}章要点：{nxt.get('summary', '')}\n情节节点：{'、'.join(nxt.get('plot_points') or [])}"
        prompt = build_outline_diverge_prompt(chapter_summary=str(cur.get("summary")), next_outline=nxt_text)
        raw = self._clean(await self._llm_chat(prompt, system_prompt="你只输出 JSON，不要输出任何其他文字。"))
        data = None
        try:
            parsed = json.loads(raw)
            if isinstance(parsed, dict):
                data = parsed
        except Exception:
            m = re.search(r"\{.*\}", raw, re.S)
            if m:
                try:
                    parsed = json.loads(m.group(0))
                    if isinstance(parsed, dict):
                        data = parsed
                except Exception:
                    pass
        if data and data.get("diverge") and str(data.get("note") or "").strip():
            return f"【动态修订提示】{str(data['note'])[:80]}"
        return ""
```
